set(LILTVIEW_RENDERER_BACKEND "OPENGL3" CACHE STRING "Render backend: OPENGL3 (D3D11 planned)")
set_property(CACHE LILTVIEW_RENDERER_BACKEND PROPERTY STRINGS OPENGL3 D3D11)
if(NOT LILTVIEW_RENDERER_BACKEND STREQUAL "OPENGL3")
    message(FATAL_ERROR "Renderer ${LILTVIEW_RENDERER_BACKEND} is not implemented yet. Use OPENGL3; Win32 + D3D11 follows the same public interface.")
endif()
find_package(OpenGL REQUIRED)
liltview_dependency_source(imgui backends/imgui_impl_opengl3.cpp imgui_source)
add_library(liltview_imgui_render_backend STATIC "${imgui_source}/backends/imgui_impl_opengl3.cpp")
target_include_directories(liltview_imgui_render_backend SYSTEM PUBLIC "${imgui_source}/backends")
target_link_libraries(liltview_imgui_render_backend PUBLIC LiltView::imgui PRIVATE OpenGL::GL ${CMAKE_DL_LIBS})

add_library(liltview_renderer STATIC validation.cpp backend/opengl/opengl_renderer.cpp)
add_library(LiltView::renderer ALIAS liltview_renderer)
target_include_directories(liltview_renderer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(liltview_renderer PUBLIC LiltView::imgui PRIVATE liltview_imgui_render_backend SDL3::SDL3-static OpenGL::GL)
liltview_configure_cxx_target(liltview_renderer)
