set(LILTVIEW_RENDERER_BACKEND "SDL_RENDER" CACHE STRING "Render backend: SDL_RENDER")
set_property(CACHE LILTVIEW_RENDERER_BACKEND PROPERTY STRINGS SDL_RENDER)
if(NOT LILTVIEW_RENDERER_BACKEND STREQUAL "SDL_RENDER")
    message(FATAL_ERROR "Only SDL_RENDER is supported. Configure with -DLILTVIEW_RENDERER_BACKEND=SDL_RENDER.")
endif()
liltview_dependency_source(imgui backends/imgui_impl_sdlrenderer3.cpp imgui_source)
add_library(liltview_imgui_render_backend STATIC "${imgui_source}/backends/imgui_impl_sdlrenderer3.cpp")
target_include_directories(liltview_imgui_render_backend SYSTEM PUBLIC "${imgui_source}/backends")
target_link_libraries(liltview_imgui_render_backend PUBLIC LiltView::imgui PRIVATE SDL3::SDL3-static)

add_library(liltview_renderer STATIC validation.cpp backend/sdl/sdl_renderer.cpp)
add_library(LiltView::renderer ALIAS liltview_renderer)
target_include_directories(liltview_renderer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(liltview_renderer PUBLIC LiltView::imgui PRIVATE liltview_imgui_render_backend SDL3::SDL3-static)
liltview_configure_cxx_target(liltview_renderer)
